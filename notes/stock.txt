### cat /opt/duer/dcslaunch.sh
#!/bin/bash
cd /opt/genie
export TZ=UTC
export HOME=/opt/genie
if [ -z "$(pidof ntpd)" ]; then
	/usr/bin/start_ntp.sh &
fi
amixer -D hw:audiocodec cset name='LINEOUT volume' 9
amixer -D hw:audiocodec cset name='MIC1 gain volume' 6.5
amixer -D hw:audiocodec cset name='MIC2 gain volume' 6.5

export GIO_EXTRA_MODULES=/opt/genie/lib/gio/modules
export LD_LIBRARY_PATH=/opt/genie/lib/pulseaudio:/opt/genie/lib/:/lib:/usr/lib
export GST_REGISTRY_UPDATE=no
export GST_PLUGIN_PATH=/opt/genie/lib/gstreamer-1.0
export GST_PLUGIN_SCANNER=/opt/genie/lib/gstreamer-1.0/gst-plugin-scanner
export XDG_CONFIG_HOME=/tmp/.config

mkdir -p /tmp/.config
if ! grep -qE '^backend=alsa' config.ini ; then
	./pulseaudio --start -v -F /opt/genie/.system.pa -p /opt/genie/lib/pulseaudio --exit-idle-time=-1 --log-target=file:/tmp/pa.log
fi

RET=1
while [ $RET -ne 0 ]; do
	if test "$1" = "--gdb" ; then
		exec /tmp/bin/gdbserver 0.0.0.0:${GDB_PORT:-1234} ./genie-client
	elif test "$1" = "--no-daemon" ; then
		exec ./genie-client
	else
		./genie-client >>/tmp/genie.log 2>&1
	fi
	RET=$?
done

exit $RET

Capture hw:audiocodec: S16/S24, 1-3 ch.  Playback hw:audiocodec: 1-2 ch.

                       ch0 MIC1   ch1 MIC2   ch2 Xadc
  silence (AC)          -55 dBFS   -55 dBFS   -94 dBFS   <- ch2 is not a microphone
  DC offset             +1.3%      +3.1%      0          <- need DC high-pass on mics
  startup click         97% peak in first ~1 ms on mics (ADC power-up); drop first 200 ms
  ch2 = MIC3 (default)  ref -41 dBFS, mics lag ref by 19-39 samples (0.4-0.8 ms)
  ch2 = DACL loopback   ref -15 dBFS (clips at 100%), mics lag ref by 39 samples
  512-tap LS echo filter, held-out chirp ERLE:  MIC3 ref 15.7/16.8 dB, DACL ref 14.3/15.2 dB

Conclusion: the "MIC3" input on the Xadc channel is a hardware tap of the
speaker/amp signal (the board's AEC reference; asound.conf calls the capture
PCM "CaptureAec"). Capture 3 ch from hw:audiocodec and use ch2 as the speex
AEC reference: it's sample-synchronous with the mics, so no delay estimation
is needed. Keep rc.local mixer settings (Xadc MIC3 Boost on, DACL off).
MIC3 gain is 0 (ref at -41 dBFS for a -20 dBFS signal); can be raised if needed.
